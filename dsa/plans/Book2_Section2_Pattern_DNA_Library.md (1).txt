━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
                          PATTERN DNA
                    DFS (Recursive & Iterative, unified card)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PARENT PATTERN
  Array / Tree / Graph Traversal

CHILDREN PATTERNS
  Tree DFS, Grid DFS, Graph DFS, Trie DFS, Backtracking DFS, Memoized DFS (Top-Down DP)

REUSABLE IDEAS
  "Explore as far as possible along each branch before backtracking", stack simulation for iterative

INPUT SHAPE
  Tree / Graph / Grid / Node

RECOGNITION SIGNALS
  "all paths", "connected components", "traverse tree/graph", "explore"

DECISION TREE
  If tree or graph traversal needed and no level info → DFS
  Else if shortest path or level → BFS

REUSABLE STATE
  visited[] or set, stack[] (implicit recursion)

FAILURE MODES
  1. Forgetting to mark visited leading to infinite loops
  2. Incorrect base case
  3. Stack overflow in deep recursion

INTERVIEW FOLLOW-UP
  How would you make it iterative? Or handle cycles?

                          PATTERN DNA
                    BFS (incl. Level-Order)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PARENT PATTERN
  Graph Traversal

CHILDREN PATTERNS
  Multi-Source BFS, Topological Sort

REUSABLE IDEAS
  "Level by level exploration using queue"

INPUT SHAPE
  Graph / Tree / Grid

RECOGNITION SIGNALS
  "shortest path in unweighted graph", "level order", "minimum steps"

DECISION TREE
  If shortest path or levels → BFS
  Else DFS for paths

REUSABLE STATE
  queue[], visited[], distance[]

FAILURE MODES
  1. Not using queue for level processing
  2. Visiting nodes prematurely
  3. Memory issues with large graphs

INTERVIEW FOLLOW-UP
  What if graph is weighted?

### Sliding Window
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
                          PATTERN DNA
                    Sliding Window
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PARENT PATTERN
  Two Pointers

CHILDREN PATTERNS
  Monotonic Deque, Variable Window + Frequency Map

REUSABLE IDEAS
  "expand-then-shrink invariant", amortized O(1) moves

INPUT SHAPE
  Array / String

RECOGNITION SIGNALS
  "contiguous subarray", "longest substring", "at most k distinct"

DECISION TREE
  If contiguous and constraint on window → Sliding Window

REUSABLE STATE
  left/right pointers, frequency map or counter

FAILURE MODES
  1. Wrong shrink condition
  2. Not updating answer at right time
  3. Off-by-one in window size

INTERVIEW FOLLOW-UP
  Minimum window or maximum with constraint?
