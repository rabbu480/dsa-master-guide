1. HASHSET - DUPLICATE CHECK
Set<Integer> set = new HashSet<>();
for (int x : nums) {
  if (!set.add(x)) return true; // already exists
}
return false;
2. HASHMAP - FREQUENCY COUNT
Map<Integer, Integer> map = new HashMap<>();
for (int x : nums) {
  map.put(x, map.getOrDefault(x, 0) + 1);
}
3. TWO SUM (HASHMAP)
Map<Integer, Integer> map = new HashMap<>();
for (int i = 0; i < nums.length; i++) {
  int need = target - nums[i];
  if (map.containsKey(need)) return new int[]{
    map.get(need), i};
  map.put(nums[i], i);
}
return new int[0];
4. GROUP ANAGRAMS (SORTED STRING KEY)
Map<String, List<String>> map = new HashMap<>();
for (String s : strs) {
  char[] arr = s.toCharArray();
  Arrays.sort(arr);
  String key = new String(arr);
  map.computeIfAbsent(key, k -> new
    ArrayList<>()).add(s);
}
return new ArrayList<>(map.values());
5. TOP K FREQUENT (BUCKET SORT)
Map<Integer, Integer> freq = new HashMap<>();
for (int x : nums) freq.put(x, freq.getOrDefault(x, 0) + 1);
List<Integer>[] bucket = new List[nums.length + 1];
for (int num : freq.keySet()) {
  int f = freq.get(num);
  if (bucket[f] == null) bucket[f] = new ArrayList<>();
  bucket[f].add(num);
}
List<Integer> res = new ArrayList<>();
for (int f = bucket.length - 1; f >= 0 && res.size() < k; f--)
  if (bucket[f] != null) res.addAll(bucket[f]);
return res.subList(0, k);
6. LONGEST CONSECUTIVE SEQUENCE
Set<Integer> set = new HashSet<>(nums);
int longest = 0;
for (int num : set) {
  if (!set.contains(num - 1)) { // start of sequence
    int cur = num, len = 1;
    while (set.contains(cur + 1)) { cur++; len++; }
    longest = Math.max(longest, len);
  }
}
return longest;

E. HASHMAP / HASHSET REMINDERS
HashMap Notes
  • map.put(key, value)
  • map.get(key)
  • map.getOrDefault(key, defaultValue)
  • map.containsKey(key)
  • map.containsValue(value)
  • map.remove(key)
  • map.size(), map.isEmpty()
  • map.keySet(), map.values(), map.entrySet()
HashSet Notes
  • set.add(x) → returns false if already exists
  • set.contains(x)
  • set.remove(x)
  • set.size(), set.isEmpty()
  • set.clear()
  • set → new ArrayList<>(set) to convert
