5.3 Backtracking Universal Template (Refer to A.22)

Mental Model: Choose → Explore → Unchoose. Search the full decision space tree.

public List<List<Integer>> subsets(int[] nums) {
    List<List<Integer>> res = new ArrayList<>();
    backtrack(0, new ArrayList<>(), nums, res);
    return res;
}

private void backtrack(int start, List<Integer> path, int[] nums, List<List<Integer>> res) {
    res.add(new ArrayList<>(path)); // Record state
    for (int i = start; i < nums.length; i++) {
        path.add(nums[i]);                  // 1. Choose
        backtrack(i + 1, path, nums, res);  // 2. Explore
        path.remove(path.size() - 1);       // 3. Unchoose
    }
}
